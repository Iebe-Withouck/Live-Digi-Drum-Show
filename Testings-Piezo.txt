      delay(40);

  }
}

***Coding part:***

```
#include "MIDIUSB.h"

struct Piezo {
  int pin;
  int note;
  int threshold;
  int peak;
  bool isHit;
  unsigned long lastHitTime;
  unsigned long noteOffTime;
};

const int numPiezos = 3;
Piezo piezos[numPiezos] = {
  {A0, 36, 60, 0, false, 0, 0}, // C1
  {A1, 38, 60, 0, false, 0, 0},  // D1
  {A2, 49, 60, 0, false, 0, 0},  // Cymbal
};

const unsigned long lockoutTime = 20;
const unsigned long noteDuration = 30;
const int minVelocity = 30;
const int maxVelocity = 127;

void setup() {
  Serial.begin(115200);
}

void loop() {
  unsigned long now = millis();

  for (int i = 0; i < numPiezos; i++) {
    handlePiezo(piezos[i], now);
  }
}

void handlePiezo(Piezo &p, unsigned long now) {
  int value = analogRead(p.pin);

  if (value > p.peak) p.peak = value;

  if (value > p.threshold && !p.isHit && now - p.lastHitTime > lockoutTime) {
    p.isHit = true;
    p.peak = value;
  }

  if (p.isHit && value < p.threshold / 2) {
    int velocity = map(p.peak, p.threshold, 900, minVelocity, maxVelocity);
    velocity = constrain(velocity, minVelocity, maxVelocity);

    noteOn(p.note, velocity);
    p.noteOffTime = now + noteDuration;

    p.isHit = false;
    p.peak = 0;
    p.lastHitTime = now;
  }

  if (p.noteOffTime != 0 && now >= p.noteOffTime) {
    noteOff(p.note);
    p.noteOffTime = 0;
  }
}

void noteOn(byte pitch, byte velocity) {
  midiEventPacket_t event = {0x09, 0x99, pitch, velocity};
  MidiUSB.sendMIDI(event);
  MidiUSB.flush();
}

void noteOff(byte pitch) {
  midiEventPacket_t event = {0x08, 0x89, pitch, 0};
  MidiUSB.sendMIDI(event);
  MidiUSB.flush();
}

```

I added a third sensor on port A2 and gave that the note of a standard cymbal so now i have a kick on piezo sensor 1 in port A0 and a snare on piezo sensor 2 in port A1 and a cymbal on piezo sensor 3 in port A2. It works! ***Coding part:*** 

```
#include "MIDIUSB.h"

const int muxAnalogPin = A0;
const int s0 = 2, s1 = 3, s2 = 4, s3 = 5;

const byte NUM_DRUMS = 2;
struct Drum {
  byte channel;
  byte note;
  int threshold;
  int peak;
  bool isHit;
  unsigned long lastHitTime;
  unsigned long minTimeBetweenHits;
  int lastPeakValue;
};

Drum drums[NUM_DRUMS] = {
  {1,  34, 250, 0, false, 0, 50, 0},
  {14, 36, 250, 0, false, 0, 50, 0}
};

const unsigned long CROSS_TALK_WINDOW = 35;
const float CROSS_TALK_RATIO = 0.3;

void setup() {
  pinMode(s0, OUTPUT);
  pinMode(s1, OUTPUT);
  pinMode(s2, OUTPUT);
  pinMode(s3, OUTPUT);
  Serial.begin(9600);
}

void loop() {
  for (int i = 0; i < NUM_DRUMS; i++) {
    selectMuxChannel(drums[i].channel);
    delayMicroseconds(800);
    handlePiezo(muxAnalogPin, drums[i], i);
  }
  delay(2);
}

void selectMuxChannel(byte channel) {
  digitalWrite(s0, channel & 0x01);
  digitalWrite(s1, (channel >> 1) & 0x01);
  digitalWrite(s2, (channel >> 2) & 0x01);
  digitalWrite(s3, (channel >> 3) & 0x01);
}

void handlePiezo(int pin, Drum &drum, int drumIndex) {
  int value = analogRead(pin);
  unsigned long currentTime = millis();

  if (value > drum.peak) {
    drum.peak = value;
  }

  if (value > drum.threshold && !drum.isHit) {
    if (currentTime - drum.lastHitTime >= drum.minTimeBetweenHits) {
      bool allowHit = true;
      
      for (int i = 0; i < NUM_DRUMS; i++) {
        if (i != drumIndex) {
          unsigned long timeSinceOtherHit = currentTime - drums[i].lastHitTime;
          
          if (timeSinceOtherHit < CROSS_TALK_WINDOW) {
            if (value < drums[i].lastPeakValue * CROSS_TALK_RATIO) {
              allowHit = false;
              break;
            }
          }
        }
      }
      
      if (allowHit) {
        drum.isHit = true;
        drum.peak = value;
      }
    }
  }

  if (drum.isHit && value < drum.threshold / 2) {
    int velocity = map(drum.peak, drum.threshold, 900, 40, 127);
    velocity = constrain(velocity, 40, 127);

    noteOn(drum.note, velocity);
    delay(5);
    noteOff(drum.note);

    drum.isHit = false;
    drum.lastPeakValue = drum.peak;
    drum.lastHitTime = currentTime;
    drum.peak = 0;
  }
}

void noteOn(byte pitch, byte velocity) {
  midiEventPacket_t event = {0x09, 0x90, pitch, velocity};
  MidiUSB.sendMIDI(event);
  MidiUSB.flush();
}

void noteOff(byte pitch) {
  midiEventPacket_t event = {0x08, 0x80, pitch, 0};
  MidiUSB.sendMIDI(event);
  MidiUSB.flush();
}

1.2 ****‼️ Going to place the sensor on the outside of “the inside of the top batter head” for better vibration/hits and no ringing, etc…

***Coding part:*** 

```
#include "MIDIUSB.h"

const int snarePin  = A0;
const int snareNote = 34;

const int threshold = 2;
const int retriggerTime = 18;

unsigned long lastTriggerTime = 0;
unsigned long peakStartTime = 0;

int peak = 0;
bool measuringPeak = false;

void setup() {
}

void loop() {
  handlePiezoAcoustic(snarePin, snareNote);
}

void handlePiezoAcoustic(int pin, int note) {
  int value = analogRead(pin);
  unsigned long now = millis();

  if (!measuringPeak &&
      value > threshold &&
      (now - lastTriggerTime) > retriggerTime) {

    measuringPeak = true;
    peak = value;
    peakStartTime = now;
  }

  if (measuringPeak) {
    if (value > peak) peak = value;

    if (now - peakStartTime >= 2) {

      int velocity = map(peak, threshold, 500, 15, 127);
      velocity = constrain(velocity, 15, 127);

      noteOn(note, velocity);
      noteOff(note);

      lastTriggerTime = now;
      measuringPeak = false;
      peak = 0;
    }
  }
}

void noteOn(byte pitch, byte velocity) {
  midiEventPacket_t event = {0x09, 0x90, pitch, velocity};
  MidiUSB.sendMIDI(event);
  MidiUSB.flush();
}

void noteOff(byte pitch) {
  midiEventPacket_t event = {0x08, 0x80, pitch, 0};
  MidiUSB.sendMIDI(event);
  MidiUSB.flush();
}

  pinMode(s1, OUTPUT);
  pinMode(s2, OUTPUT);
  pinMode(s3, OUTPUT);
}

void loop() {
  int maxValue = 0;
  int maxIndex = -1;
  unsigned long now = millis();

  for (int i = 0; i < NUM_DRUMS; i++) {
    selectMuxChannel(drums[i].channel);
    delayMicroseconds(1000);

    int value = analogRead(muxAnalogPin);

    if (value > drums[i].peak) {
      drums[i].peak = value;
    }

    if (value > drums[i].threshold &&
        (now - drums[i].lastHitTime) > drums[i].minTimeBetweenHits) {

      if (value > maxValue) {
        maxValue = value;
        maxIndex = i;
      }
    }
  }

  if (maxIndex != -1) {
    Drum &d = drums[maxIndex];

    int velocity = map(d.peak, d.threshold, 900, 40, 127);
    velocity = constrain(velocity, 40, 127);

    noteOn(d.note, velocity);
    delay(5);
    noteOff(d.note);

    d.lastHitTime = now;
  }

  for (int i = 0; i < NUM_DRUMS; i++) {
    if (now - drums[i].lastHitTime > HIT_WINDOW) {
      drums[i].peak = 0;
    }
  }

  delay(1);
}

void selectMuxChannel(byte channel) {
  digitalWrite(s0, channel & 0x01);
  digitalWrite(s1, (channel >> 1) & 0x01);
  digitalWrite(s2, (channel >> 2) & 0x01);
  digitalWrite(s3, (channel >> 3) & 0x01);
}

void noteOn(byte pitch, byte velocity) {
  midiEventPacket_t event = {0x09, 0x90, pitch, velocity};
  MidiUSB.sendMIDI(event);
  MidiUSB.flush();
}

void noteOff(byte pitch) {
  midiEventPacket_t event = {0x08, 0x80, pitch, 0};
  MidiUSB.sendMIDI(event);
  MidiUSB.flush();
}
