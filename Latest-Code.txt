struct Piezo {
  byte muxChannel;
  int threshold;
  int peak;
  bool isHit;
  unsigned long lastHitTime;
  unsigned long noteOffTime;
};

const int minVelocity = 40;
const int maxVelocity = 127;
const unsigned long lockoutTime = 40;

void setup() {
  Serial.begin(115200);

  pinMode(PIEZO_S0, OUTPUT);
  pinMode(PIEZO_S1, OUTPUT);
  pinMode(PIEZO_S2, OUTPUT);
  pinMode(PIEZO_S3, OUTPUT);

  pinMode(POT_S0, OUTPUT);
  pinMode(POT_S1, OUTPUT);
  pinMode(POT_S2, OUTPUT);
  pinMode(POT_S3, OUTPUT);
}

void loop() {
  unsigned long now = millis();

  int pot1Value = readPotMux(pot1Channel);
  int pot2Value = readPotMux(pot2Channel);
  int pot3Value = readPotMux(pot3Channel);
  int pot4Value = readPotMux(pot4Channel);
  int pot5Value = readPotMux(pot5Channel);
  int pot6Value = readPotMux(pot6Channel);
  int pot7Value = readPotMux(pot7Channel);
  int pot8Value = readPotMux(pot8Channel);
  int pot9Value = readPotMux(pot9Channel);
  int pot10Value = readPotMux(pot10Channel);
  int pot11Value = readPotMux(pot11Channel);

  int zone1 = constrain(pot1Value / 64, 0, 15);
  int zone2 = constrain(pot2Value / 64, 0, 15);
  int zone3 = constrain(pot3Value / 64, 0, 15);
  int zone4 = constrain(pot4Value / 64, 0, 15);
  int zone5 = constrain(pot5Value / 64, 0, 15);
  int zone6 = constrain(pot6Value / 64, 0, 15);
  int zone7 = constrain(pot7Value / 64, 0, 15);
  int zone8 = constrain(pot8Value / 64, 0, 15);
  int zone9 = constrain(pot9Value / 64, 0, 15);
  int zone10 = constrain(pot10Value / 64, 0, 15);
  int zone11 = constrain(pot11Value / 64, 0, 15);

  handlePiezo(piezo1, now, track1Notes[zone1], 1);
  handlePiezo(piezo2, now, track2Notes[zone2], 2);
  handlePiezo(piezo3, now, track3Notes[zone3], 3);
  handlePiezo(piezo4, now, track4Notes[zone4], 4);
  handlePiezo(piezo5, now, track5Notes[zone5], 5);
  handlePiezo(piezo6, now, track6Notes[zone6], 6);
  handlePiezo(piezo7, now, track7Notes[zone7], 7);
  handlePiezo(piezo8, now, track8Notes[zone8], 8);
  handlePiezo(piezo9, now, track9Notes[zone9], 9);
  handlePiezo(piezo10, now, track10Notes[zone10], 10);
  handlePiezo(piezo11, now, track11Notes[zone11], 11);

  delayMicroseconds(300);
}

int readPiezoMux(byte channel) {
  digitalWrite(PIEZO_S0, bitRead(channel, 0));
  digitalWrite(PIEZO_S1, bitRead(channel, 1));
  digitalWrite(PIEZO_S2, bitRead(channel, 2));
  digitalWrite(PIEZO_S3, bitRead(channel, 3));
  delayMicroseconds(5);
  return analogRead(PIEZO_MUX_SIG);
}

int readPotMux(byte channel) {
  digitalWrite(POT_S0, bitRead(channel, 0));
  digitalWrite(POT_S1, bitRead(channel, 1));
  digitalWrite(POT_S2, bitRead(channel, 2));
  digitalWrite(POT_S3, bitRead(channel, 3));
  delayMicroseconds(5);
  return analogRead(POT_MUX_SIG);
}

void handlePiezo(Piezo &p, unsigned long now, byte midiNote, byte channel) {
  int value = readPiezoMux(p.muxChannel);

  if (value > p.peak) p.peak = value;

  if (value > p.threshold && !p.isHit && now - p.lastHitTime > lockoutTime) {
    p.isHit = true;
    p.peak = value;
  }

  if (p.isHit && value < p.threshold / 2) {
    int velocity = map(p.peak, p.threshold, 900, minVelocity, maxVelocity);
    velocity = constrain(velocity, minVelocity, maxVelocity);

    noteOn(midiNote, velocity, channel);

    p.isHit = false;
    p.peak = 0;
    p.lastHitTime = now;
  }
}

void noteOn(byte pitch, byte velocity, byte channel) {
  midiEventPacket_t event = {0x09, 0x90 + (channel - 1), pitch, velocity};
  MidiUSB.sendMIDI(event);
  MidiUSB.flush();
}

void noteOff(byte pitch, byte channel) {
  midiEventPacket_t event = {0x08, 0x80 + (channel - 1), pitch, 0};
  MidiUSB.sendMIDI(event);
  MidiUSB.flush();
}
