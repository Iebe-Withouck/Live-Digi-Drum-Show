#include "MIDIUSB.h"

const int muxAnalogPin = A0;
const int s0 = 2, s1 = 3, s2 = 4, s3 = 5;
const int threshold = 100;

const byte NUM_DRUMS = 2;
struct Drum {
  byte channel;
  byte note;
  int peak;
  bool isHit;
};

Drum drums[NUM_DRUMS] = {
  {1,  34, 0, false},
  {14, 36, 0, false}
};

void setup() {
  pinMode(s0, OUTPUT);
  pinMode(s1, OUTPUT);
  pinMode(s2, OUTPUT);
  pinMode(s3, OUTPUT);
  Serial.begin(9600);
}

void loop() {
  for (int i = 0; i < NUM_DRUMS; i++) {
    selectMuxChannel(drums[i].channel);
    delayMicroseconds(50);
    handlePiezo(muxAnalogPin, drums[i].note, drums[i].peak, drums[i].isHit);
  }
  delay(1);
}

void selectMuxChannel(byte channel) {
  digitalWrite(s0, channel & 0x01);
  digitalWrite(s1, (channel >> 1) & 0x01);
  digitalWrite(s2, (channel >> 2) & 0x01);
  digitalWrite(s3, (channel >> 3) & 0x01);
}

void handlePiezo(int pin, int note, int &peak, bool &isHit) {
  int value = analogRead(pin);

  if (value > peak) {
    peak = value;
  }

  if (value > threshold && !isHit) {
    isHit = true;
    peak = value;
  }

  if (isHit && value < threshold / 2) {
    int velocity = map(peak, threshold, 900, 40, 127);
    velocity = constrain(velocity, 40, 127);

    noteOn(note, velocity);
    delay(5);
    noteOff(note);

    isHit = false;
    peak = 0;
  }
}

void noteOn(byte pitch, byte velocity) {
  midiEventPacket_t event = {0x09, 0x90, pitch, velocity};
  MidiUSB.sendMIDI(event);
  MidiUSB.flush();
}

void noteOff(byte pitch) {
  midiEventPacket_t event = {0x08, 0x80, pitch, 0};
  MidiUSB.sendMIDI(event);
  MidiUSB.flush();
}
