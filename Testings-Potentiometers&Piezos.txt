# First Testing Demo with Potentiometers

1. **wired-up one potentiometer to one piezo element to my arduino board and try to change the sounds when turning the potentiometer from the piezo vibrations, following the Arduino documentation for potentiometers:** https://docs.arduino.cc/learn/electronics/potentiometer-basics/

***Coding part:*** 

```
#include "MIDIUSB.h"

struct Piezo {
  int pin;
  int threshold;
  int peak;
  bool isHit;
  unsigned long lastHitTime;
  unsigned long noteOffTime;
};

Piezo piezo = {A0, 15, 0, false, 0, 0};

const int potPin = A3;

const int numSounds = 16;
const byte soundNotes[numSounds] = {
  36,  // C2
  32,  // G#1
  28,  // E1
  24,  // C1
  25,  // C#2
  33,  // A1
  29,  // F1
  39,  // D#2
  38,  // D2
  34,  // A#1
  30,  // F#1
  26,  // D1
  27,  // D#1
  54,  // F#3
  47,  // B1
  31   // G1
};

const unsigned long lockoutTime = 25;
const unsigned long noteDuration = 30;
const int minVelocity = 50;
const int maxVelocity = 127;

int lastZone = -1;

void setup() {
  Serial.begin(115200);
}

void loop() {
  unsigned long now = millis();

  int potValue = analogRead(potPin);

  int currentZone = map(potValue, 0, 1015, 0, numSounds - 1);
  currentZone = constrain(currentZone, 0, numSounds - 1);

  Serial.print("Pot value: ");
  Serial.print(potValue);
  Serial.print("  |  Selected MIDI note: ");
  Serial.println(soundNotes[currentZone]);

  handlePiezo(piezo, now, soundNotes[currentZone]);

  delay(20);
}

void handlePiezo(Piezo &p, unsigned long now, byte midiNote) {
  int value = analogRead(p.pin);

  if (value > p.peak) p.peak = value;

  if (value > p.threshold && !p.isHit && now - p.lastHitTime > lockoutTime) {
    p.isHit = true;
    p.peak = value;
  }

  if (p.isHit && value < p.threshold / 2) {
    int velocity = map(p.peak, p.threshold, 900, minVelocity, maxVelocity);
    velocity = constrain(velocity, minVelocity, maxVelocity);

    noteOn(midiNote, velocity);
    p.noteOffTime = now + noteDuration;

    p.isHit = false;
    p.peak = 0;
    p.lastHitTime = now;
  }

  if (p.noteOffTime != 0 && now >= p.noteOffTime) {
    noteOff(midiNote);
    p.noteOffTime = 0;
  }
}

void noteOn(byte pitch, byte velocity) {
  midiEventPacket_t event = {0x09, 0x99, pitch, velocity};
  MidiUSB.sendMIDI(event);
  MidiUSB.flush();
}

void noteOff(byte pitch) {
  midiEventPacket_t event = {0x08, 0x89, pitch, 0};
  MidiUSB.sendMIDI(event);
  MidiUSB.flush();
}
```

I assigned different pot values to different notes so when i change the potentiometers value, there is a different value sent to Arduino IDE, in Arduino IDE there are different notes assigned to different values, and than Arduino IDE sends that MIDI note to Logic Pro.   digitalWrite(PIEZO_S0, bitRead(channel, 0));
  digitalWrite(PIEZO_S1, bitRead(channel, 1));
  digitalWrite(PIEZO_S2, bitRead(channel, 2));
  digitalWrite(PIEZO_S3, bitRead(channel, 3));
  delayMicroseconds(5);
  return analogRead(PIEZO_MUX_SIG);
}

int readPotMux(byte channel) {
  digitalWrite(POT_S0, bitRead(channel, 0));
  digitalWrite(POT_S1, bitRead(channel, 1));
  digitalWrite(POT_S2, bitRead(channel, 2));
  digitalWrite(POT_S3, bitRead(channel, 3));
  delayMicroseconds(5);
  return analogRead(POT_MUX_SIG);
}

void handlePiezo(Piezo &p, unsigned long now, byte midiNote, byte channel) {
  int value = readPiezoMux(p.muxChannel);

  if (value > p.peak) p.peak = value;

  if (value > p.threshold && !p.isHit && now - p.lastHitTime > lockoutTime) {
    p.isHit = true;
    p.peak = value;
  }

  if (p.isHit && value < p.threshold / 2) {
    int velocity = map(p.peak, p.threshold, 900, minVelocity, maxVelocity);
    velocity = constrain(velocity, minVelocity, maxVelocity);

    noteOn(midiNote, velocity, channel);

    p.isHit = false;
    p.peak = 0;
    p.lastHitTime = now;
  }
}

void noteOn(byte pitch, byte velocity, byte channel) {
  midiEventPacket_t event = {0x09, 0x90 + (channel - 1), pitch, velocity};
  MidiUSB.sendMIDI(event);
  MidiUSB.flush();
}

void noteOff(byte pitch, byte channel) {
  midiEventPacket_t event = {0x08, 0x80 + (channel - 1), pitch, 0};
  MidiUSB.sendMIDI(event);
  MidiUSB.flush();
}
